	\section{ مقدمه و مفاهیم اصلی \GLS{VINS}}
	\subsection{مسئله: ردیابی حرکت}
	\gls{VINS} یک چارچوب ادغام حسگر است که برای تخمین موقعیت سه‌بعدی، جهت‌گیری و سرعت یک عامل متحرک
	(که همه‌ی این‌ها با هم وضعیت یا \emph{Pose}) استفاده می‌شود. ما نیاز داریم در هر ثانیه چندین بار این اطلاعات را بدست آوردیم تا درمورد هدایت و جهت‌گیری عامل تصمیم‌گیری کنیم.
	
	\subsection{ادغام حسگرها}
	دو حسگر ارزان و متداول می‌توانند کمک کنند، اما هر کدام ضعف‌های بزرگ دارند:
	\begin{itemize}
		\item \textbf{دوربین:} مانند چشم‌ انسان عمل می‌کند. دنیا را می‌بیند و می‌تواند با مشاهده حرکت ویژگی‌ها (مانند گوشه‌های یک پنجره) در تصویر، تشخیص دهد که \emph{حرکت کرده است}. با این حال، به تنهایی نمی‌تواند تشخیص دهد \emph{چقدر} حرکت کرده است.
		این مسئله \textbf{ابهام مقیاس (\lr{Scale Ambiguity})} نام دارد. همچنین در تاریکی یا اگر تصویر بیش از حد تار باشد، عملکردش مختل می‌شود
		\cite{huang2019visual}.
		
		اگر چه دوربین اطلاعات غنی و با وضوح بالا درباره محیط فراهم می‌کند. با ردیابی چگونگی تغییر ظاهر محیط، می‌تواند حرکت را تخمین بزند. با این حال، در برابر محو شدن حرکت، بافت کم و تغییرات سریع نورپردازی آسیب‌پذیر است. به طور حیاتی، یک دوربین به تنهایی نمی‌تواند مقیاس مطلق محیط (یعنی فاصله اجسام تا چه اندازه است) را درک کند.
		\item \textbf{واحد اندازه‌گیری لختی (\GLS{IMU}):}
		مانند گوش داخلی انسان عمل می‌کند. حرکت را احساس می‌کند. یک ژیروسکوپ (سرعت چرخش را اندازه می‌گیرد) و یک شتاب‌سنج (شتاب، شامل گرانش را اندازه می‌گیرد) دارد. برای حرکات سریع و کوتاه عالی است. اما اندازه‌گیری‌های آن بسیار پُرنویز هستند. اگر سعی شود موقعیت خود را با انتگرال‌گیری از این سیگنال‌های نویزی محاسبه شود، خطاها به سرعت انباشته می‌شوند و منجر به \textbf{(Drift)} می‌شوند. در عرض چند ثانیه، موقعیت تخمین‌زده شده کاملاً اشتباه خواهد بود.
		
		معمولاً \LR{IMU} شامل یک شتاب‌سنج سه‌محوره و یک ژیروسکوپ سه‌محوره است. نیروی وارده (شامل گرانش) و سرعت زاویه‌ای را در فرکانسی بسیار بالا
		(\LR{\SI{100}{Hz}} تا \LR{\SI{1}{KHz}}) اندازه‌گیری می‌کند. این امر به آن اجازه می‌دهد تا حرکت با فرکانس بالا را ردیابی کند، اما اندازه‌گیری‌های آن با نویز و بایاس قابل توجهی مخدوش شده که منجر به رانش سریع هنگام انتگرال‌گیری می‌شود
		\cite{mourikis2007multi}.
	\end{itemize}
	
	یک \gls{VINS} نرم‌افزاری است که این دو حسگر را ادغام (Fuse) می‌کند. 	\gls{VINS} به عنوان «مغز» عمل می‌کند که این جریان‌های داده ناقص را هوشمندانه ادغام می‌کند. از دقت بلندمدت سیستم بینایی برای تصحیح رانش IMU استفاده می‌کند و از داده‌های با فرکانس بالا IMU برای پر کردن فاصله‌های بین فریم‌های کندتر دوربین، ردیابی حرکت در هنگام تخریب بینایی و حل ابهام مقیاس ذاتی در بینایی بهره می‌برد.  در کنار هم، این‌ها یک سیستم قوی و دقیق ایجاد می‌کند.

	\section{بررسی ماژول به ماژول \gls{VINS}}
	\subsection{پیش‌پردازش حسگر: پاک‌سازی داده‌ها}
	داده‌های خام حسگر نویزی هستند. این ماژول آن‌ها را پاک‌سازی می‌کند.
	\begin{itemize}
		\item \textbf{دوربین:} 
		تصویر توسط لنز در حرکت دچار اعوجاج
		(\LR{Distortion})
		می‌شود (خطوط صاف به صورت منحنی به نظر می‌رسند).
		این اتفاق در اثر فاصله‌ی زمانی بین نمونه‌برداری از خازن‌های حساس به نور دوربین اتفاق می‌افتد.
		\emph{اعوجاج‌زدایی} را با استفاده از پارامترهای ذاتی دوربین (فاصله کانونی $f_x, f_y$، نقطه اصلی $c_x, c_y$) و ضرایب اعوجاج  برای معکوس کردن اثرات لنز اعمال می‌کند.
		\item \textbf{\LR{IMU}:} حسگر دارای خطاهای کوچک داخلی به نام \emph{بایاس (Biases)} است. از مقادیر کالیبراسیون اولیه برای حذف این خطاهای شناخته‌شده استفاده می‌کنیم \cite{rehder2016kalibr}.
		\item \textbf{همگام‌سازی:}
		حیاتی‌ترین وظیفه است. برچسب‌های زمانی دوربین و \gls{IMU} باید هم‌راستا شوند. این اغلب شامل تخمین یک افست زمانی ثابت $t_d$ بین دو کلاک است. عدم همگام‌سازی مستقیماً به خطا در موقعیت تخمین زده شده تبدیل می‌شود
		\cite{qin2018online}.
	\end{itemize}
	
	\subsection{مقداردهی اولیه}
	نمی‌شود سیستم اصلی را از صفر شروع کرد. به یک حدس اولیه برای موارد زیر نیاز است:
	\begin{itemize}
		\item جهت گرانش.
		\item سرعت حرکت.
		\item مقیاس دنیای واقعی (یک حرکت پیکسل نشان‌دهنده چند متر است).
		\item بایاس‌های \gls{IMU}.
	\end{itemize}
	ماژول مقداردهی اولیه از چند ثانیه اول داده برای حل این مسئله استفاده می‌کند. نحوه حرکت ویژگی‌ها در دوربین و نحوه احساس گرانش و حرکت توسط \LR{IMU} را مشاهده می‌کند و سپس یک مسئله کوچک را برای همتراز کردن این دو اندازه‌گیری و به دست آوردن مقادیر اولیه محاسبه می‌کند.
	
	
	\subsection{پیش‌پرداز بینایی: ردیاب ویژگی}
	این بخش بینایی کامپیوتر است. وظیفه آن پیدا کردن اشیایی مانند گربه یا ماشین نیست، بلکه پیدا کردن و ردیابی \emph{نقاط} متمایز در تصویر، مانند گوشه یک کتاب یا یک نقطه روی یک فرش بافت‌دار، است. به این نقاط \textbf{ویژگی (Features)} می‌گویند.
	\begin{enumerate}
		\item \textbf{تشخیص:}  از الگوریتم‌هایی مانند
		\LR{FAST}، شی-توماسی یا \LR{ORB} برای آشکارسازی \emph{نقاط کلیدی} برجسته در یک تصویر استفاده می‌شود. ویژگی‌های خوب، گوشه‌ها یا نواحی با کنتراست بالا هستند که ردیابی آن‌ها آسان است.
		\item \textbf{ردیابی:} برای تصویر بعدی، سعی می‌کند همان نقاط را دوباره پیدا کند. از الگوریتم‌هایی مانند جریان نوری \gls{KLT} برای این کار استفاده می‌شود. با مشاهده چگونگی حرکت یک ویژگی از \lr{(x1, y1)} به \lr{(x2, y2)}، سرنخی از چگونگی حرکت دوربین به دست می‌آوریم.  الگوریتم جریان نوری  \gls{KLT} ابزار استاندارد برای ردیابی این ویژگی‌ها در فریم‌های متوالی است. این الگوریتم خطای فوتومتری (تفاوت در شدت‌های پیکسل) یک وصله حول نقطه کلیدی را کمینه می‌کند.
		\item \textbf{حذف داده‌های پرت:} گاهی ردیابی اشتباه می‌کند. از الگوریتم‌هایی مانند \gls{RANSAC} برای شناسایی و دور ریختن این ردیف‌های نادرست استفاده می‌شود. .
	\end{enumerate}
	خروجی، یک لیست قابل اعتماد از نقاط مطابقت‌یافته بین تصاویر متوالی است. 
	
	\subsection{انتگرال‌گیری اینرسی: شبیه‌ساز فیزیک}
	این ماژول داده‌های IMU را مدیریت می‌کند. این یک شبیه‌سازی فیزیک ساده است:
	\begin{enumerate}
		\item از آخرین موقعیت، سرعت و جهت گیری شناخته شده شروع می‌کند.
		\item هر بار که یک خوانش جدید IMU دریافت می‌شود (مثلاً هر \lr{\SI{5}{ms}})، چنین محاسبه می‌شود:
		\begin{quote}
			\textit{«از آنجایی که شما \textbf{اینجا} بودید، و با این نرخ \textbf{در حال چرخش} و با این نرخ \textbf{شتاب گیری} بودید در \lr{\SI{5}{ms}} گذشته ، باید \textbf{اکنون} \textbf{اینجا} باشید.»}
		\end{quote}
		\item این کار را با \textbf{انتگرال‌گیری}  از سرعت زاویه‌ای برای به دست آوردن چرخش و یکبار انتگرال‌گیری از شتاب  (نیرو) برای  به دست آوردن برآیند تغییر سرعت و دو بار برای به دست آوردن  برای تغییر موقعیت انجام می‌دهیم.
	\end{enumerate}
	این روش بسیار ساده و سریع است اما به دلیل خطاهای کوچک، تقریباً بلافاصله از واقعیت منحرف می‌شود 
	(\lr{Drift}
	می‌کند). فقط برای پیش‌بینی حرکت در زمان کوتاه بین فریم‌های دوربین مفید است.
	
	\subsubsection{مدل ریاضی برای \LR{IMU}}
	این مؤلفه با استفاده از \gls{IMU} \emph{حدس مسیر} را انجام می‌دهد. هسته آن مدل سینماتیکی IMU است. وضعیت سامانه به صورت زیر تعریف می‌شود:
	\[
	\mathbf{x} = \left[ \mathbf{p}^w, \mathbf{q}^w_b, \mathbf{v}^w, \mathbf{b}_a, \mathbf{b}_g \right]^T
	\]
	که در آن $\mathbf{p}^w$ موقعیت در چارچوب جهان، $\mathbf{q}^w_b$ چهارگان نمایش‌دهنده جهت‌گیری از چارچوب جسم به جهان، $\mathbf{v}^w$ سرعت در چارچوب جهان و $\mathbf{b}_a$ و $ \mathbf{b}_g$ به ترتیب بایاس‌های شتاب‌سنج و ژیروسکوپ هستند \cite{forster2015imu}.
	
	مدل دینامیک زمان-پیوسته به صورت زیر است:
	\begin{align*}
		\dot{\mathbf{p}}^w &= \mathbf{v}^w \\
		\dot{\mathbf{v}}^w &= \mathbf{R}^w_b (\mathbf{a}_m - \mathbf{b}_a - \mathbf{n}_a) + \mathbf{g}^w \\
		\dot{\mathbf{q}}^w_b &= \frac{1}{2} \mathbf{q}^w_b \otimes \boldsymbol{\omega}_m - \mathbf{b}_g - \mathbf{n}_g \\
		\dot{\mathbf{b}}_a &= \mathbf{n}_{ba} \\
		\dot{\mathbf{b}}_g &= \mathbf{n}_{bg}
	\end{align*}
	که در آن $\mathbf{a}_m$ و $\boldsymbol{\omega}_m$ اندازه‌گیری‌های خام IMU هستند، $\mathbf{n}_a, \mathbf{n}_g, \mathbf{n}_{ba}, \mathbf{n}_{bg}$ جمله‌های نویز هستند، $\mathbf{R}^w_b$ ماتریس چرخش از چارچوب جسم به جهان است و $\mathbf{g}^w = [0, 0, -9.81]^T$ گرانش است.
	
	برای استفاده کارآمد از این اندازه‌گیری‌ها در یک بهینه‌ساز زمان-گسسته، از تئوری \textbf{پیش‌انتگرال‌گیری IMU} استفاده می‌شود. اندازه‌گیری‌های IMU بین دو لحظه زمانی $i$ و $j$ در \emph{چارچوب بدنی محلی} $i$ انتگرال‌گیری می‌شوند که قیود نسبی $\Delta \mathbf{R}_{ij}, \Delta \mathbf{v}_{ij}, \Delta \mathbf{p}_{ij}$ را به دست می‌دهد که مستقل از وضعیت اولیه $\mathbf{x}_i$ و بایاس‌های آن هستند. سپس این اندازه‌گیری پیش‌انتگرال‌گیری شده به عنوان یک قید بین وضعیت‌های $i$ و $j$ در بهینه‌ساز استفاده می‌شود \cite{forster2016manifold}.
	\subsubsection{شبه‌کد محاسبه موقعیت و سرعت}
	\begin{latin}
		\begin{lstlisting}[caption={\rl{شبه کد انتگرال‌گیری از واحد اندازه‌گیری اینرسی}}, label={lst:imupseudo}]
			# This function runs for every new IMU measurement
			def integrate_imu(imu_data, current_state):
			# dt is the time since the last IMU measurement
			dt = imu_data.timestamp - current_state.timestamp
			
			# 1. Update orientation by integrating gyroscope reading
			delta_angle = imu_data.gyro * dt
			new_orientation = current_state.orientation * delta_angle
			
			# 2. Update velocity by integrating acceleration
			# Note: We must rotate acceleration into the world frame and remove gravity
			acceleration_world = new_orientation.rotate(imu_data.accel)
			acceleration_world -= gravity_vector # [0, 0, 9.8] m/s^2
			new_velocity = current_state.velocity + acceleration_world * dt
			
			# 3. Update position by integrating velocity
			new_position = current_state.position + new_velocity * dt
			
			# 4. Update the state with new values
			current_state = State(new_position, new_velocity, new_orientation)
			return current_state
		\end{lstlisting}
	\end{latin}
	\subsection{بهینه‌سازی پسین: تصمیم‌گیری نهایی}
	این مغز سیستم است، جایی که ادغام اتفاق می‌افتد. به هر دو حسگر توجه می‌کند:
	\begin{enumerate}
		\item \textbf{پیش‌پرداز بینایی}
		می‌گوید: «فکر می‌کنم دوربین اینطور حرکت کرده است زیرا دیدم این ویژگی‌ها حرکت کردند.»
		\item \textbf{انتگرال‌گیر اینرسی}
		می‌گوید: «فکر می‌کنم دوربین اینطور حرکت کرده است زیرا این نیروها را احساس کردم.»
		
		
	\end{enumerate}
	وظیفه بهینه‌ساز این است که محتمل‌ترین سناریو را پیدا کند که متناسب  با هر دو ورودی باشد. سپس به طور پیوسته تخمین‌های خود از وضعیت‌های گذشته را تنظیم می‌کند تا خطاهای هم دوربین و هم IMU را به حداقل برساند. این فرآیند، IMU را \textbf{تصحیح} می‌کند \cite{qin2018vins}.
	
	بهینه‌سازی هسته‌ی \gls{VINS} است که به عنوان یک مسئله تخمین \Gls{MAP} فرمول‌بندی می‌شود. این مؤلفه محتمل‌ترین دنباله از وضعیت‌ها $\mathcal{X} = \{\mathbf{x}_0, \mathbf{x}_1, ..., \mathbf{x}_n\}$ را با توجه به همه مشاهدات $\mathcal{Z}$ (اندازه‌گیری‌های IMU و ردیابی‌های ویژگی تصویری) پیدا می‌کند \cite{indelman2012factor}.
	\[
	\mathcal{X}^* = \arg\max_{\mathcal{X}} P(\mathcal{X} | \mathcal{Z})
	\]
	
	این معمولاً با استفاده از کمینه‌سازی مربعات غیرخطی حل می‌شود. تابع هزینه کلی $J(\mathcal{X})$ مجموعی از جمله‌های خطای مربع است:
	\[
	J(\mathcal{X}) = \sum_{k} \lVert \mathbf{e}_{\text{imu},k} \rVert^2_{\mathbf{\Sigma}_k} + \sum_{l} \lVert \mathbf{e}_{\text{vision},l} \rVert^2_{\mathbf{\Lambda}_l}
	\]
	که در آن:
	\begin{itemize}
		\item $\mathbf{e}_{\text{imu},k}$ خطای اندازه‌گیری پیش‌انتگرال‌گیری شده IMU بین دو وضعیت است.
		\item $\mathbf{e}_{\text{vision},l}$ \emph{خطای بازتابش} برای یک ویژگی $l$ مشاهده شده در چندین فریم است. اگر موقعیت سه‌بعدی تخمین زده شده ویژگی $l$، $\lambda_l$ باشد و در پیکسل $(u, v)$ در فریمی مشاهده شود که پیکسل پیش‌بینیشده در آن $\pi(\mathbf{T}_{cw}, \lambda_l)$ است (با استفاده از مدل دوربین $\pi$)، خطا برابر است با $\mathbf{e}_{\text{vision}} = (u, v) - \pi(\mathbf{T}_{cw}, \lambda_l)$.
		\item $\mathbf{\Sigma}_k$ و $\mathbf{\Lambda}_l$ ماتریس‌های کوواریانس نمایش‌دهنده عدم قطعیت اندازه‌گیری‌های مربوطه هستند.
	\end{itemize}
	
	برای مدیریت پیچیدگی محاسباتی، از یک رویه \textbf{پنجره لغزان} استفاده می‌شود. بهینه‌ساز فقط تعداد ثابتی از وضعیت‌های اخیر را در پنجره نگه می‌دارد. هنگامی که یک وضعیت جدید اضافه می‌شود، قدیمی‌ترین وضعیت با استفاده از مکمل شور \emph{حاشیه‌ای می‌شود}. این کار تمام اطلاعات حاصل از اندازه‌گیری‌های مربوط به آن وضعیت قدیمی را به یک پیشین روی وضعیت‌های باقی‌مانده تبدیل می‌کند، که اندازه مسئله را محدود نگه می‌دارد در حالی که اطلاعات را حفظ می‌کند.

	\section{سیستم عامل ربات
		(\LR{ROS})}
	\label{sec:ros}
	
	\textbf{سیستم عامل ربات
		\gls{ROS}
	} یک فریم‌ورک یا چارچوب نرم‌افزاری برای توسعه برنامه‌های رباتیکی است. اگرچه نام آن حاوی کلمه «سیستم عامل» است، اما 
	\gls{ROS} یک سیستم عامل سنتی مانند ویندوز یا لینوکس نیست، بلکه مجموعه‌ای از ابزارها، کتابخانه‌ها و قراردادها 
	\LR{(Conventions)} است که هدف آن ساده‌سازی ساخت برنامه‌های پیچیده و قوی برای ربات‌ها در انواع پلتفرم‌های سخت‌افزاری است. هستهٔ اصلی 
	\gls{ROS} ارائه زیرساختی برای سرویس‌های توزیع‌شده 
	است که به گره‌های 
	مختلف اجازه می‌دهد تا از طریق یک پروتکل ارتباطی سبک‌وزن (مانند انتشار/اشتراک‌گذاری پیام‌ها یا سرویس‌ها) با یکدیگر تعامل داشته باشند. این معماری مبتنی بر گره، امکان ماژولار بودن، قابلیت استفاده مجدد کدها و اجرای همزمان وظایف را فراهم می‌کند.
	
	کاربردهای اصلی \gls{ROS} شامل موارد زیر است:
	\begin{itemize}
		\item \textbf{ابزارهای سطح پایین}:
		ارائه کنترل‌کننده‌های سخت‌افزاری، درایورهای سنسورها (مانند دوربین، 
		\LR{IMU}، لیدار) و الگوریتم‌های پایه.
		\item \textbf{ارسال پیام‌ها}:
		فراهم آوردن یک مکانیسم ارتباطی قوی برای تبادل داده بین پردازش‌های مختلف.
		\item \textbf{ابزارهای سطح بالا}:
		ارائه ابزارهای بینایی مانند \textbf{RViz} برای نمایش داده‌های سنسورها و حالت ربات، و ابزارهای اشکال‌زدایی.
		\item \textbf{بسته‌های نرم‌افزاری}:
		وجود هزاران بسته  از پیش توسعه‌یافته برای پیاده‌سازی الگوریتم‌های متداول در بینایی کامپیوتر، ناوبری، نقشه‌برداری همزمان و مکان‌یابی (SLAM) و پردازش داده‌های سنسوری.
	\end{itemize}
	
	\subsection{کاربرد \Gls{ROS} در سیستم‌های ناوبری بصری-اینرسیال (VINS)}
	\label{subsec:ros_in_vins}
	
	در حوزه
	\GLS{VINS}، \gls{ROS} 
	نقش یک چسب نرم‌افزاری حیاتی را ایفا می‌کند. این سیستم‌ها برای تخمین حالت (موقعیت و جهت) ربات با ادغام داده‌های یک دوربین (داده بصری)
	و یک واحد اندازه‌گیری اینرسی
	(\gls{IMU}) طراحی شده‌اند. \gls{ROS} بستری ایده‌آل برای توسعه و آزمایش این الگوریتم‌های پیچیده فراهم می‌آورد.
	\paragraph{مزایای کلیدی استفاده از \gls{ROS}  در \gls{VINS} عبارتند از:}
	
	\begin{itemize}
		\item \textbf{یکپارچه‌سازی سنسورها}: با استفاده از درایورهای از پیش ساخته‌شده \gls{ROS} برای دوربین‌ها و \LR{IMU}ها، می‌توان به سرعت داده‌های خام این سنسورها را دریافت و در قالب پیام‌های استاندارد (مانند \texttt{sensor\_msgs/Image} و \texttt{sensor\_msgs/Imu}) منتشر کرد.
		\item \textbf{ماژولاریتی}: الگوریتم VINS را می‌توان به چندین گره مستقل تقسیم کرد (مثلاً یک گره برای پیش‌پردازش تصویر، یک گره برای ادغام داده‌ها و یک گره برای تخمین حالت). این امر توسعه، تست و بهینه‌سازی هر بخش را به صورت مجزا ممکن می‌سازد.
		\item \textbf{بینایی (Visualization)}: ابزارهایی مانند \textbf{RViz} امکان نمایش بلادرنگ تخمین مسیر \LR{(Trajectory)}، ویژگی‌های استخراج‌شده از تصاویر، نقاط ابری و داده‌های IMU را فراهم می‌کنند که برای درک رفتار الگوریتم و اشکال‌زدایی آن ضروری است.
		\item \textbf{ضبط و پخش داده (Rosbag)}: با استفاده از ابزار \texttt{rosbag} می‌توان داده‌های سنسورها را به راحتی ضبط (\textit{record}) و بعداً برای آزمایش و مقایسه الگوریتم‌ها در شرایط یکسان، مجدداً پخش (\textit{playback}) کرد. این قابلیت برای توسعه الگوریتم‌های VINS که وابستگی زیادی به داده دارند، بسیار ارزشمند است.
		\\
		به کمک این امکان، دیتاست 
		\LR{EuRoC MAV} 
		ضبط شده است که ما از آن استفاده خواهیم کرد.
		\item \textbf{استانداردسازی و جامعه بزرگ}: استفاده از \gls{ROS} تضمین می‌کند که کد توسعه‌یافته با طیف وسیعی از سخت‌افزارها و نرم‌افزارهای دیگر سازگار است و می‌توان از هزاران بسته و منبع موجود در جامعه بزرگ کاربران 
		\gls{ROS} بهره برد.
	\end{itemize}
	
	\subsection{مراحل نصب \gls{ROS}}
	\label{subsec:ros_installation}
	
	نصب \gls{ROS} معمولاً بر روی سیستم عامل اوبونتو 
	\LR{(Ubuntu)} انجام می‌شود. در حال حاضر برای نسخه 
	\lr{\textbf{24.04}} اوبونتو
	، بسته‌های 
	\textit{دبیان}
	رسمی
	\LR{ \textbf{\gls{ROS} 2 }}
	(مانند 
	\lr{\gls{ROS} 2 Jazzy}) در دسترس هستند. مراحل نصب به صورت خلاصه به شرح زیر است:
	
	\begin{enumerate}
		\item \textbf{پیکربندی محیط}:
		\begin{itemize}
			\item اطمینان از تنظیم صحیح \LR{Locale} سیستم.
			\item افزودن مخازن (\LR{Repository}) رسمی \gls{ROS} به سیستم.
			\item به روزرسانی کش مخازن: 
			\begin{LTR}
				\begin{verbatim}sudo apt update\end{verbatim}
			\end{LTR}
		\end{itemize}
		
		\item \textbf{نصب نسخه دسکتاپ}: این نسخه شامل ابزارهای بینایی مانند \LR{RViz} و دموها است.
		\begin{LTR}
			\begin{verbatim}sudo apt install \gls{ROS}-jazzy-desktop\end{verbatim}
		\end{LTR}
		
		
		\item \textbf{تنظیم محیط}:
		پس از نصب، باید فایل \texttt{setup.bash} مربوطه در هر ترمینال سورس (Source) شود تا دستورات \gls{ROS} در دسترس قرار گیرند:
		\begin{LTR}
			\begin{verbatim}source /opt/ros/jazzy/setup.bash\end{verbatim}
		\end{LTR}
		اگر این خط به فایل \texttt{\textasciitilde/.bashrc} اضافه شود تا به صورت خودکار در ترمینال‌ها اجرا می‌گردد.
		
		\item \textbf{تست نصب}:
		با اجرای یک گره انتشارکننده (\textit{talker}) و یک گره دریافت‌کننده (\textit{listener}) می‌توان صحت نصب را بررسی کرد. این معادل 
		\LR{\textit{Hello World}}  در این سیستم می‌باشد.
	\end{enumerate}
	
	
	\section{بررسی \gls{VINS}‌های موجود}
	در این حوزه پروژه‌های متنوعی توسعه یافته‌اند که هر یک بر پایه‌ی رویکردهای متفاوتی طراحی شده‌اند. با جستجو  درمورد 
	\gls{VINS}
	به سه پروژه‌ی متن‌-باز محبوب‌تر از دیدگاه کاربران رسیدیم. سه برنامه‌ی محبوب‌تر در پلتفرم گیت‌هاب بر اساس تعداد ستاره‌ها:
	\begin{latin}
		\begin{enumerate}
			\item \textbf{\href{https://github.com/HKUST-Aerial-Robotics/VINS-Mono}{VINS-Mono}:} 5.5K $\bigstar$
			\item \textbf{\href{https://github.com/HKUST-Aerial-Robotics/VINS-Fuesion}{VINS-Fusion}:} 4.1K $\bigstar$
			\item \textbf{\href{https://github.com/rpng/open_vins}{OpenVINS}:} 2.5K $\bigstar$
		\end{enumerate}
	\end{latin}
	
	از بین پروژه‌های موجود، پروژه‌ی 
	\LR{VINS-Mono} کلاسیک‌ترین پروژه است که در آن از الگوریتم‌های غیرخطی برای تخمین و از روش‌های کلاسیک در پردازش تصویر  یک دوربینی و پردازش \gls{IMU} بهره برده شده. 
	\LR{VINS-Fusion} بر پایه‌ی
	\LR{VINS-Mono} ساخته شده و دارای قابلیت‌های بیشتری از 
	\LR{VINS-Mono}
	است. پشتیبانی از پردازش‌های پیچیده‌تر و چند دوربینی به این سیستم افزوده شده. پروژه‌ی 
	\LR{OpenVINS} توسط یک تیم دیگر توسعه یافته که بروزترین تکنولوژی‌های موجود 
	(\LR{state-of-the-art}) در آن به کار برده شده است.
	
	
	\subsection{VINS-Mono}
	\LR{VINS-Mono}
	یک سیستم \gls{VINS}
	مبتنی بر \textbf{بهینه‌سازی غیرخطی} است که پایه‌ی آن بر رویکرد \textbf{فکتورگراف‌ها}
	(\LR{Factor Graphs})
	و \textbf{بهینه‌سازی پنجره‌ی لغزان} 
	(\LR{Sliding Window Optimization})
	استوار است
	\cite{qin2018vins}.
	این پروژه به‌طور خاص برای سیستم‌های تک‌دوربینه (\LR{Monocular})
	طراحی شده و قابلیت‌های اصلی آن عبارتند از:
	\begin{itemize}
		\item تخمین دقیق وضعیت (\LR{Pose})، سرعت و بایاس سنسورها با استفاده از داده‌های دوربین و \gls{IMU}
		\item پشتیبانی از بستن حلقه‌ها (\LR{Loop Closure}) برای اصلاح خطای انباشت شده
		\item قابلیت بازسازی نقشه‌ی سه‌بعدی متراکم
		\item عملکرد بلادرنگ (\RL{Real-time}) به کمک پردازش موازی
		\item پیاده‌سازی مبتنی بر\gls{ROS}  و 
		\RL{C++}
		با استفاده از کتابخانه‌ی 
		\LR{\verb|ceres-solver| \lr{1.14.0}} 
	\end{itemize}
	این سیستم برای محیط‌های داخلی و خارجی با بافت‌های غنی عملکرد مطلوبی دارد، اما در محیط‌های فاقد بافت یا با حرکت‌های سریع ممکن است دچار چالش شود.
	
	\subsection{\LR{VINS-Fusion}}
	\LR{VINS-Fusion}
	به‌عنوان نسخه‌ی توسعه‌یافته‌ی
	\LR{VINS-Mono}
	طراحی شده و پایه‌ی آن همان \textbf{بهینه‌سازی غیرخطی} با قابلیت \textbf{هم‌جوشی چندحسی }
	(\LR{Multi-sensor Fusion}) است. این پروژه با هدف پشتیبانی از سنسورهای متنوع توسعه یافته و شامل ویژگی‌های زیر است:
	\begin{itemize}
		\item 
		پشتیبانی از پیکربندی‌های مختلف سنسوری: تک‌دوربینه
		(مونوکولار)، دودوربینه (استریو)، دوربین‌های عمق‌سنج 
		(\LR{RGB-D}) و \LR{GPS}
		\item هم‌جوشی داده‌های بصری، اینرسی و 
		\LR{GPS}
		برای افزایش دقت و قابلیت اطمینان
		\item قابل استفاده در محیط‌های بزرگ و خارجی با استفاده از داده‌های
		\LR{GPS}
		برای محدود کردن خطای انباشت شده
		\item بهبود عملکرد در محیط‌های کم‌بافت با استفاده از داده‌های استریو یا عمق
		\item سازگاری با سخت‌افزارهای متنوع از جمله پهپادها و ربات‌های متحرک
	\end{itemize}
	این سیستم با بهره‌گیری از معماری ماژولار، امکان افزودن سنسورهای جدید را فراهم می‌کند و برای کاربردهای نیازمند دقت بالا در مقیاس‌های بزرگ مناسب است.
	
	\subsection{OpenVINS}
	\LR{OpenVINS} یک سیستم تخمین حالت 
	\gls{VINS}
	مبتنی بر \textbf{فیلتر کالمن} است که پایه‌ی آن بر رویکرد 
	\gls{MSCKF}
	استوار است
	\cite{geneva2019openvins}.
	این پروژه با تمرکز بر بهینه‌سازی محاسباتی و سادگی پیاده‌سازی طراحی شده و ویژگی‌های کلیدی آن عبارتند از:
	\begin{itemize}
		\item استفاده از فیلتر کالمن به‌جای بهینه‌سازی غیرخطی برای کاهش پیچیدگی محاسباتی
		\item پشتیبانی از حالت‌های تک‌دوربینه و استریو با قابلیت انتخاب‌پذیری ماژول‌ها
		\item عملکرد بلادرنگ با مصرف منابع محاسباتی پایین (مناسب برای سیستم‌های نهفته
		یا \LR{Embedded})
		\item پیاده‌سازی مدرن با استفاده از Eigen و OpenCV و بدون وابستگی به \gls{ROS}
		\item قابلیت گسترش آسان برای افزودن ویژگی‌های جدید مانند هم‌جوشی
		GPS
		\item مستندات جامع و تست‌های واحد گسترده برای اطمینان از پایداری
	\end{itemize}
	این سیستم برای کاربردهایی که نیاز به پاسخگویی سریع و مصرف بهینه‌ی منابع دارند (مانند عامل‌های کوچک با منابع محدود و پهپادهای سبک) انتخاب مناسبی است، اما ممکن است در سناریوهای بسیار پیچیده دقتی کمتر از روش‌های مبتنی بر بهینه‌سازی داشته باشد.

	\section{بیلدگرفتن و اجرا کردن \LR{VINS-Mono}}
	همه‌ی سیستم‌های \gls{VINS} برای اجرا نیازمند سیستم‌عامل و محیط مشخصی هستند و همین موضوع، بیلد گرفتن و اجرا گرفتن از این پروژه‌ها را سخت می‌کند. تلاش‌های اولیه‌ی ما برای بیلد گرفتن از هر سه پروژه روی 
	\lr{ubuntu 24.04 (LTS)} 
	به همراه
	\lr{\gls{ROS} 2 Jazzy (LTS)}
	بی‌ثمر بود.
	در نهایت، با توجه به نیازمندی تحقیقاتی ما و هدف این کارآموزی که آشنایی با 
	\gls{VINS}
	بود ما 
	\LR{VINS-Mono} 
	را برای پیاده‌سازی انتخاب کردیم. زیرا از نظر سازگاری و اجرا بهترین گزینه بود. همچنین دیتاست‌های موجود 
	(مثلا 
	\LR{EuRoC MAV})
	بر پایه‌ی
	\lr{\gls{ROS} 1}
	ساخته شده بودند.
	
	\subsection{محیط موردنیاز}
	\LR{VINS-Mono} برای 
	\lr{ubuntu 16.04 (LTS)} نوشته شده است
	و با نسخه‌ی
	\lr{\gls{ROS} 1 Kinetic} سازگار است.
	آماده‌سازی محیط برای اجرای 
	\LR{VINS-Mono}  شامل سه مرحله می‌شود. مرحله‌ی اول 
	نصب سیستم‌عامل اوبونتو در محیط مجازی، مرحله‌ی دوم نصب کتابخانه‌های موردنیاز و مرحله‌ی سوم بیلد گرفتن از پروژه می‌باشد.
	
	\subsubsection{نصب اوبونتو به صورت مجازی}
	ما از محیط 
	\lr{qemu} به کمک ابزار مجازی‌سازی
	(\rl{Virtualization})
	\lr{KVM} استفاده می‌کنیم.
	برای نصب پکیج‌های مورد نیاز از دستور زیر استفاده می‌کنیم.
	\begin{LTR}
		\begin{verbatim}sudo apt install qemu-kvm libvirt-daemon-system \
			libvirt-clients bridge-utils virtinst\end{verbatim}
	\end{LTR}
	سپس یک دیسک مجازی با فضای کافی می‌سازیم.
	\begin{LTR}
		\begin{verbatim}qemu-img create -f qcow2 /var/lib/libvirt/images/ubuntuـdisk.qcow2 20G\end{verbatim}
	\end{LTR}
	و در مرحله‌ی بعد به کمک 
	\LR{\verb|virt-install|}
	دیسک اوبونتو را داخل دیسک مجازی نصب می‌کنیم.
	
	\begin{LTR}
		\begin{verbatim}virt-install \
			--name ubuntu \
			--ram 4096 \
			--vcpus 2 \
			--disk path=/var/lib/libvirt/images/ubuntu_disk.qcow2,format=qcow2 \
			--cdrom ~/images/ubuntu-16.04.7-desktop-amd64.iso \
			--os-variant ubuntu16.04 \
			--network bridge=virbr0 \
			--graphics spice\end{verbatim}
	\end{LTR}
	
	بعد از نصب، راه‌اندازی و اجرای محیط مجازی به‌سادگی با دستور زیر امکان پذیر خواهد بود.
	
	\begin{LTR}
		\begin{verbatim}qemu-system-x86_64 \
			-m 4096 \
			-smp 4 \
			-drive file=ubuntu_disk.qcow2,format=qcow2 \
			-display sdl\end{verbatim}
	\end{LTR}
	اما مشکلاتی در اجرا به این صورت هست:
	\begin{itemize}
		\item صفحه‌نمایش به صورت adaptive نیست و ثابت است.
		\item سیستم‌عامل بسیار کند و laggy است.
	\end{itemize}
	
	راه‌حل استفاده و بهره‌گیری از امکانات مجازی‌سازی KVM مثل شتاب‌دهنده‌ی سخت‌افزاری 
	(\LR{Hardware Acceleration})
	است. بنابرین دستور اجرای کمو را به صورت زیر تغییر می‌دهیم.
	\begin{LTR}
		\begin{verbatim}qemu-system-x86_64 \
			-enable-kvm \
			-cpu host \
			-m 4096 \
			-smp 4 \
			-drive file=ubuntu-16.04.qcow2,format=qcow2,if=virtio \
			-vga virtio \
			-display sdl,gl=on \
			-usb -device usb-tablet
		\end{verbatim}
	\end{LTR}

	\subsubsection{نصب \Gls{ROS} نسخه‌ی \rl{Kinetic}}
	به کمک 
	\href{https://wiki.ros.org/kinetic/Installation/Ubuntu}{مستندات پروژه}
	نسخه‌ی Kinetic را روی اوبونتو نصب می‌کنیم.
	
	\begin{LTR}
		\begin{verbatim}sudo apt-get install ros-kinetic-desktop
		\end{verbatim}
	\end{LTR}
	سپس برای سادگی در استفاده مجدد، متغیرهای محیطی 
	\gls{ROS}

	\subsubsection{نصب پکیج‌های موردنیاز 
		\LR{VINS-Mono}}
	پکیچ‌های مورد نیاز برای \LR{VINS-Mono} شامل پکیج‌های زیر می‌باشد.
	\begin{LTR}
		\begin{verbatim}
			sudo apt install ros-kinetic-cv-bridge \
			ros-kinetic-tf ros-kinetic-message-filters ros-kinetic-im
		\end{verbatim}
	\end{LTR}
	
	همچنین برای بیلد گرفتن و اجرای این پروژه، نیاز به کتابخانه‌ی 
	\LR{verb|ceres-solver|} داریم. متاسفانه نسخه‌ی موجود در سیستم مدیریت بسته‌های اوبونتو 
	(apt) نسخه‌ی جدیدتری از آن را دارد 
	(نسخه‌ی ۲.۲) که  \LR{VINS-Mono} با آن سازگاری ندارد. به همین‌جهت ما باید از 
	\href{https://github.com/ceres-solver/ceres-solver/releases/tag/1.14.0}{مخزن گیت‌هاب}
	\LR{\verb|ceres-solver|} 
	دقیقا کد منبع نسخه‌ی 
	\lr{1.14.0} را استخراج کنیم و به صورت دستی بیلد بگیریم.
	اول به نصب کتابخانه‌های موردنیاز شروع می‌کنیم.
	\begin{LTR}
		\begin{verbatim}
			sudo apt install \
			cmake build-essential \
			libblas-dev liblapack-dev libsuitesparse-dev  \
			libgoogle-glog-dev libgflags-dev \
			libatlas-base-dev  libeigen3-dev
		\end{verbatim}
	\end{LTR}
	پس از نصب پکیج‌های مورد نیاز، در یک دایرکتوری موقتی کرس را دانلود می‌کنیم و سپس از آن بیلد می‌گیریم.
	\begin{LTR}
		\begin{verbatim}
			wget https://github.com/ceres-solver/ceres-solver/archive/refs/tags/1.14.0.zip
			tar 1.14.0.zip
			cd 1.14.0
			mkdir build
			cd build
			cmake ..
			make -j4
			sudo make install
		\end{verbatim}
	\end{LTR}

	\section{تشخیص و ردیابی ویژگی‌های تصویری در VINS-Mono}
	
	تشخیص و ردیابی ویژگی‌ها، پایه و اساس بینایی کامپیوتری در VINS-Mono را تشکیل می‌دهند و با شناسایی و پیگیری نقاط متمایز تصویر در فریم‌های متوالی، مسافت‌پیمایی 
	(Odommetry)
	تصویری-اینرسیِ مقاوم را امکان‌پذیر می‌سازند. در این بخش، مبانی نظری الگوریتم‌های تشخیص و ردیابی ویژگی‌های به‌کاررفته در VINS-Mono را معرفی می‌کنیم و بر روش تشخیص گوشه‌های شی-توماسی و الگوریتم ردیابی جریان نوری کاناده-لوکاس-توماسی 
	\gls{KLT} \ref{kltsection}
	تمرکز دارد. درک این تکنیک‌ها برای تحلیل و اصلاح کد منبع VINS-Mono ضروری است، زیرا این الگوریتم‌ها مستقیماً بر دقت سیستم، کارایی محاسباتی و مقاومت آن در محیط‌های متنوع تأثیر می‌گذارند.
	
	\subsection{تشخیص ویژگی‌های مناسب: تشخیص گوشه‌های شی-توماسی}
	
	\subsubsection{مبانی نظری}
	VINS-Mono از الگوریتم تشخیص گوشه‌های شی-توماسی 
	(\LR{Shi-Tomasi Corner Detection}) 
	برای
	شناسایی «ویژگی‌های مناسب برای ردیابی» در هر فریم تصویری استفاده می‌کند. این روش، که توسط شی و توماسی معرفی شده است \cite{shi1994good}، بهبود یافته‌ای بر تشخیص‌دهنده‌های گوشه‌ی سنتی است و به‌ویژه برای کاربردهای ردیابی مناسب می‌باشد.
	
	بینش اصلی تشخیص شی-توماسی این است که یک ویژگی مناسب برای ردیابی باید در چندین فریم قابل ردیابی باشد. الگوریتم، کیفیت نقاط بالقوه ویژگی را با استفاده از فرمول‌بندی ریاضی زیر ارزیابی می‌کند:
	
	برای یک نقطه تصویری داده‌شده $(x,y)$، الگوریتم ماتریس گرادیان $2 \times 2$ به نام $M$ را می‌سازد:
	\begin{equation*}
		M = \sum_{x,y} w(x,y) \begin{bmatrix}
			I_x^2 & I_x I_y \\
			I_x I_y & I_y^2
		\end{bmatrix}
	\end{equation*}
	که در آن $I_x$ و $I_y$ گرادیان‌های تصویری در جهت‌های $x$ و $y$ هستند و $w(x,y)$ تابع وزنی (معمولاً پنجره گاوسی) است.
	
	تابع پاسخ گوشه شی-توماسی 
	(\LR{Corner Response Function})
	به صورت زیر تعریف می‌شود:
	\begin{equation*}
		R = \min(\lambda_1, \lambda_2)
	\end{equation*}
	که در آن $\lambda_1$ و $\lambda_2$ مقادیر ویژه ماتریس $M$ هستند. یک نقطه، در صورتی گوشه مناسبی تلقی می‌شود که $R$ از آستانه از پیش تعریف‌شده‌ای فراتر رود. این معیار با روش تشخیص گوشه‌ی‌ سنتی، که از $\lambda_1 + \lambda_2 - k(\lambda_1 \lambda_2)$ به عنوان تابع پاسخ استفاده می‌کند، متفاوت است.
	
	\subsubsection{پیاده‌سازی عملی در VINS-Mono}
	در
	\LR{VINS-Mono}،
	تشخیص‌دهنده شی-توماسی با در نظر گرفتن چندین ملاحظه عملی پیاده‌سازی شده است:
	
	\begin{itemize}
		\item \textbf{توزیع یکنواخت}: الگوریتم، توزیع یکنواخت ویژگی‌ها را در سراسر تصویر با حفظ حداقل فاصله جداسازی بین گوشه‌های تشخیص‌داده‌شده تضمین می‌کند. این امر از تجمع ویژگی‌ها جلوگیری کرده و پوشش فضایی مناسبی برای تخمین موقعیت مقاوم فراهم می‌آورد.
		
		\item \textbf{آستانه‌گذاری تطبیقی}: آستانه تشخیص اغلب بر اساس محتوای تصویر و شرایط نوری تطبیق داده می‌شود تا تعداد ویژگی‌های ثابت در محیط‌های مختلف حفظ شود.
		
		\item \textbf{ماسک‌گذاری ویژگی}: مناطق پویا (مانند آسمان یا اشیاء متحرک) ممکن است ماسک شوند تا از تشخیص ویژگی‌های ناپایدار جلوگیری شود.
	\end{itemize}
	
	روش شی-توماسی در 
	\LR{VINS-Mono}
	ترجیح داده می‌شود زیرا مستقیماً برای قابلیت ردیابی بهینه‌سازی شده و برای مرحله ردیابی بعدی ایده‌آل است.
	
	\subsection{ردیابی ویژگی: جریان نوری \GLS{KLT}}
	\label{kltsection}
	پس از تشخیص ویژگی‌های مناسب،
	\LR{VINS-Mono} آن‌ها را در فریم‌های متوالی با استفاده از الگوریتم  \gls{KLT} \ ردیابی می‌کند که بر پایه روش جریان نوری لوکاس-کاناده اصلی \cite{lucas1981iterative} بنا شده است.
	
	روش \gls{KLT} فرض ثبات روشنایی و حرکت کوچک بین فریم‌ها را می‌پذیرد. برای یک نقطه ویژگی در موقعیت $(x,y)$ در فریم $t$، الگوریتم به دنبال یافتن جابه‌جایی $(dx, dy)$ آن در فریم $t+1$ است به گونه‌ای که:
	\begin{equation*}
		I(x,y,t) = I(x+dx, y+dy, t+1)
	\end{equation*}
	
	با استفاده از بسط تیلور مرتبه اول، این امر به معادله محدودیت جریان نوری منجر می‌شود:
	\begin{equation}
		I_x \cdot v_x + I_y \cdot v_y = -I_t
	\end{equation}
	که در آن $v_x$ و $v_y$ مؤلفه‌های سرعت هستند و $I_t$ گرادیان زمانی است.
	
	برای یک همسایگی از پیکسل‌ها، این سیستم بیش‌تعیین‌شده‌ای را ایجاد می‌کند:
	\begin{equation}
		\begin{bmatrix}
			I_x(p_1) & I_y(p_1) \\
			I_x(p_2) & I_y(p_2) \\
			\vdots & \vdots \\
			I_x(p_n) & I_y(p_n)
		\end{bmatrix}
		\begin{bmatrix}
			v_x \\
			v_y
		\end{bmatrix}
		=
		\begin{bmatrix}
			-I_t(p_1) \\
			-I_t(p_2) \\
			\vdots \\
			-I_t(p_n)
		\end{bmatrix}
	\end{equation}
	
	حل آن با استفاده از کمترین مربعات به دست می‌آید:
	\begin{equation}
		\begin{bmatrix}
			v_x \\
			v_y
		\end{bmatrix}
		= (A^T A)^{-1} A^T b
	\end{equation}
	
	\subsubsection{بهبودهای ردیاب \gls{KLT}}
	توماسی و کاناده \cite{tomasi1991detection} روش \gls{KLT} را گسترش دادند تا ردیاب ویژگی مقاومی با چندین بهبود کلیدی ایجاد کنند:
	
	\begin{itemize}
		\item \textbf{پالایش تکراری}: ردیاب KLT از روش تکراری  
		(\LR{Itarative}) 
		برای پالایش موقعیت ویژگی استفاده می‌کند و دقت را برای حرکات بزرگ‌تر بهبود می‌بخشد.
		
		\item \textbf{نظارت بر کیفیت ویژگی}: الگوریتم، کیفیت ویژگی را در حین ردیابی به طور مداوم با استفاده از همان معیار به‌کاررفته در تشخیص (حداقل مقدار ویژه ماتریس گرادیان) نظارت می‌کند. ویژگی‌هایی که ضعیف شناخته می‌شوند، حذف می‌گردند.
		
		\item \textbf{پیاده‌سازی هرمی}: برای مدیریت حرکات بزرگ‌تر، ردیاب KLT بر روی هرم‌های تصویری عمل می‌کند و از سطوح رزولوشن درشت به ریز ردیابی می‌نماید.
	\end{itemize}
	
	\subsection{ادغام در VINS-Mono}
	
	
	\LR{VINS-Mono}
	یک استراتژی مدیریت ویژگی پیچیده را پیاده‌سازی می‌کند که تعادل بین تشخیص و ردیابی را برقرار می‌سازد:
	
	\begin{enumerate}
		\item \textbf{تشخیص اولیه}: هنگامی که سیستم شروع می‌شود یا عملکرد ردیابی کاهش می‌یابد، ویژگی‌های جدید با استفاده از روش شی-توماسی تشخیص داده می‌شوند.
		
		\item \textbf{ردیابی مداوم}: ویژگی‌های موجود با استفاده از الگوریتم \gls{KLT} در فریم‌های متوالی ردیابی می‌گردند.
		
		\item \textbf{حذف ویژگی}: ویژگی‌هایی که ردیابی آن‌ها شکست می‌خورد (به دلیل پوشیدگی، تاری حرکت یا خروج از میدان دید) حذف شده و با تشخیص‌های جدید جایگزین می‌شوند.
		
		\item \textbf{بررسی اختلاف 
			(\LR{Parallex})}: ویژگی‌هایی با پارالاکس کافی (اختلاف ناشی از حرکت دوربین) برای تخمین موقعیت اولویت‌بندی می‌شوند، زیرا محدودیت‌های هندسی قوی‌تری فراهم می‌آورند.
	\end{enumerate}
	
	\subsubsection{ملاحظات محاسباتی}
	پیاده‌سازی در VINS-Mono برای عملکرد بلادرنگ بهینه‌سازی شده است:
	
	\begin{itemize}
		\item \textbf{مجموعه ویژگی‌های پراکنده}: به جای پردازش تمام پیکسل‌ها، تنها یک مجموعه پراکنده از ویژگی‌های باکیفیت بالا (معمولاً ۱۰۰-۲۰۰ پیکسل) ردیابی می‌شود که بار محاسباتی را کاهش می‌دهد.
		
		\item \textbf{پردازش موازی}: تشخیص و ردیابی ویژگی می‌تواند در چندین رشته CPU موازی‌سازی شود.	
		\item \textbf{تعداد ویژگی تطبیقی}: تعداد ویژگی‌های ردیابی‌شده بر اساس کیفیت ردیابی و منابع محاسباتی موجود به طور پویا تنظیم می‌شود.
	\end{itemize}
	
	
	برای ویرایش و بهینه‌سازی پروژه‌ی 
	\LR{VINS-Mono}
	نیازمند شناخت دقیق و با جزئیات هر کدام از این قسمت‌هاست:
	
	\begin{itemize}
		\item \textbf{تنظیم پارامترها}: رفتار VINS-Mono می‌تواند به طور قابل توجهی توسط پارامترهایی مانند آستانه‌های تشخیص ویژگی، حداقل جداسازی ویژگی و اندازه‌های پنجره ردیابی تحت تأثیر قرار گیرد.
		
		\item \textbf{بهینه‌سازی عملکرد}: دانش پیچیدگی محاسباتی این الگوریتم‌ها به بهینه‌سازی عملکرد سیستم برای محدودیت‌های سخت‌افزاری خاص کمک می‌کند.
		
		\item \textbf{افزایش مقاومت در برابر خطا}: درک معیارهای کیفیت ویژگی، توسعه استراتژی‌های مدیریت ویژگی مقاوم‌تر برای محیط‌های چالش‌برانگیز (با بافت کم، صحنه‌های پویا و غیره) را امکان‌پذیر می‌سازد.
		
		\item \textbf{اصلاح الگوریتم}: طراحی ماژولار VINS-Mono اجازه جایگزینی الگوریتم‌های تشخیص یا ردیابی ویژگی جایگزین را در صورت مناسب بودن برای کاربردهای خاص می‌دهد.
	\end{itemize}
	
	الگوریتم‌های شی-توماسی و 
	\gls{KLT}
	، VINS-Mono را با یک ابزار قابل اعتماد و کارآمد برای استخراج ویژگی‌های بصری مجهز می‌کنند و پایه‌ای برای بهینه‌سازی تصویری-اینرسی بعدی فراهم می‌آورند که تخمین حالت دقیق و مقاوم  به خطا را امکان‌پذیر می‌سازد.

	\subsection{مدیریت پارامترها}
	\label{subsec:parameter_management}
	
	فایل \texttt{parameters.cpp} مسئول خواندن پارامترهای پیکربندی از فایل YAML است. این پارامترها شامل موارد زیر هستند:
	
	\begin{itemize}
		\item \textbf{پارامترهای ROS}: \texttt{IMAGE\_TOPIC} و \texttt{IMU\_TOPIC} برای تعیین تاپیک‌های ورودی
		\item \textbf{پارامترهای ردیابی ویژگی}: 
		\begin{itemize}
			\item \texttt{MAX\_CNT}: حداکثر تعداد ویژگی‌ها برای ردیابی
			\item \texttt{MIN\_DIST}: حداقل فاصله بین ویژگی‌ها
			\item \texttt{F\_THRESHOLD}: آستانه برای حذف نقاط پرت با ماتریس بنیادی
			\item \texttt{FREQ}: فرکانس انتشار ویژگی‌ها
		\end{itemize}
		\item \textbf{پارامترهای تصویر}: \texttt{ROW} و \texttt{COL} برای ابعاد تصویر
		\item \textbf{پارامترهای دوربین}: \texttt{FISHEYE} برای مشخص کردن نوع دوربین
	\end{itemize}
	
	تابع \texttt{readParameters} پارامترها را از فایل پیکربندی می‌خواند و مقادیر پیش‌فرض را برای پارامترهایی که در فایل وجود ندارند تنظیم می‌کند. این تابع همچنین مسیر فایل ماسک برای دوربین‌های ماهی‌چشمی را تنظیم می‌کند.
	
	\subsection{گره اصلی ROS}
	\label{subsec:main_ros_node}
	
	فایل \texttt{feature\_tracker\_nodes.cpp} گره اصلی ROS را پیاده‌سازی می‌کند که وظایف زیر را انجام می‌دهد:
	
	\subsubsection{پردازش تصاویر ورودی}
	\label{subsubsec:image_processing}
	
	تابع \texttt{img\_callback} برای هر تصویر ورودی فراخوانی می‌شود و مراحل زیر را انجام می‌دهد:
	
	\begin{enumerate}
		\item \textbf{تشخیص ناپیوستگی در جریان تصاویر}: اگر بین تصاویر متوالی فاصله زمانی بیش از حد زیاد باشد، سیستم را بازنشانی می‌کند.
		\item \textbf{کنترل فرکانس}: با استفاده از شمارنده \texttt{pub\_count} و زمان شروع، فرکانس انتشار ویژگی‌ها را کنترل می‌کند.
		\item \textbf{تبدیل تصویر}: تصویر را به فرمت خاکستری (\lr{MONO8}) تبدیل می‌کند.
		\item \textbf{بهبود کنتراست}: اگر پارامتر \texttt{EQUALIZE} فعال باشد، از الگوریتم CLAHE برای بهبود کنتراست تصویر استفاده می‌کند.
		\item \textbf{ردیابی ویژگی}: تابع \texttt{readImage} از کلاس \texttt{FeatureTracker} را برای ردیابی ویژگی‌ها فراخوانی می‌کند.
	\end{enumerate}
	
	\subsubsection{انتشار ویژگی‌ها}
	\label{subsubsec:feature_publishing}
	
	پس از پردازش تصویر، ویژگی‌های ردیابی شده به صورت یک پیام \texttt{sensor\_msgs::PointCloud} منتشر می‌شوند. این پیام شامل اطلاعات زیر است:
	
	\begin{itemize}
		\item موقعیت‌های نرمال‌شده ویژگی‌ها در مختصات تصویر
		\item شناسه منحصر به فرد هر ویژگی
		\item مختصات پیکسلی ویژگی‌ها
		\item سرعت ویژگی‌ها در تصویر
	\end{itemize}
	
	اگر پارامتر \texttt{SHOW\_TRACK} فعال باشد، تصویری با ویژگی‌های ردیابی شده نیز منتشر می‌شود.
	
	\subsection{ ردیاب ویژگی}
	\label{subsec:feature_tracker_core}
	
	کلاس \texttt{FeatureTracker} در فایل \texttt{feature\_tracker.cpp} پیاده‌سازی اصلی الگوریتم ردیابی ویژگی را شامل می‌شود.
	
	\subsubsection{ساختارهای داده}
	\label{subsubsec:data_structures}
	
	کلاس \texttt{FeatureTracker} از ساختارهای داده زیر برای ذخیره اطلاعات ویژگی‌ها استفاده می‌کند:
	
	\begin{itemize}
		\item \texttt{forw\_pts, cur\_pts, prev\_pts}: مختصات ویژگی‌ها در تصویر فعلی، قبلی و قبلی‌تر
		\item \texttt{ids}: شناسه‌های منحصر به فرد ویژگی‌ها
		\item \texttt{track\_cnt}: تعداد فریم‌هایی که هر ویژگی ردیابی شده است
		\item \texttt{cur\_un\_pts}: مختصات نرمال‌شده و بدون اعوجاج ویژگی‌ها
		\item \texttt{pts\_velocity}: سرعت ویژگی‌ها در تصویر
	\end{itemize}
	
	\subsubsection{الگوریتم ردیابی}
	\label{subsubsec:tracking_algorithm}
	
	الگوریتم اصلی ردیابی در تابع \texttt{readImage} پیاده‌سازی شده است. این تابع مراحل زیر را انجام می‌دهد:
	
	\begin{enumerate}
		\item \textbf{بهبود کنتراست}: اگر پارامتر \texttt{EQUALIZE} فعال باشد، از CLAHE برای بهبود کنتراست تصویر استفاده می‌کند.
		
		\item \textbf{ردیابی ویژگی‌های موجود}: با استفاده از جریان نوری Lucas-Kanade (\texttt{cv::calcOpticalFlowPyrLK})، ویژگی‌های موجود در تصویر قبلی را در تصویر فعلی ردیابی می‌کند.
		
		\item \textbf{حذف نقاط پرت}: با استفاده از تابع \texttt{rejectWithF}، نقاطی که با ماتریس بنیادی سازگار نیستند را حذف می‌کند.
		
		\item \textbf{ایجاد ماسک}: با تابع \texttt{setMask}، ماسکی ایجاد می‌کند تا از تشخیص ویژگی‌های جدید در نزدیکی ویژگی‌های موجود جلوگیری کند.
		
		\item \textbf{تشخیص ویژگی‌های جدید}: با استفاده از \texttt{cv::goodFeaturesToTrack}، ویژگی‌های جدید را در مناطقی که ماسک اجازه می‌دهد تشخیص می‌دهد.
		
		\item \textbf{اضافه کردن ویژگی‌های جدید}: با تابع \texttt{addPoints}، ویژگی‌های جدید را به لیست ویژگی‌های ردیابی شده اضافه می‌کند.
		
		\item \textbf{نرمال‌سازی و محاسبه سرعت}: با تابع \texttt{undistortedPoints}، مختصات ویژگی‌ها را نرمال کرده و سرعت آن‌ها را محاسبه می‌کند.
	\end{enumerate}
	
	\subsubsection{حذف نقاط پرت با ماتریس بنیادی}
	\label{subsubsec:outlier_rejection}
	
	تابع \texttt{rejectWithF} از ماتریس بنیادی برای حذف نقاط پرت استفاده می‌کند. این تابع مراحل زیر را انجام می‌دهد:
	
	\begin{enumerate}
		\item نرمال‌سازی مختصات ویژگی‌ها با استفاده از مدل دوربین
		\item محاسبه ماتریس بنیادی با استفاده از الگوریتم \gls{RANSAC}
		\item حذف ویژگی‌هایی که با ماتریس بنیادی سازگار نیستند
	\end{enumerate}
	
	این روش به حذف ویژگی‌هایی که در اثر حرکت‌های ناگهانی دوربین یا تغییرات نوری ایجاد شده‌اند کمک می‌کند.
	
	\subsubsection{مدیریت ویژگی‌ها}
	\label{subsubsec:feature_management}
	
	تابع \texttt{setMask} ماسکی برای مدیریت توزیع ویژگی‌ها در تصویر ایجاد می‌کند. این تابع:
	
	\begin{itemize}
		\item ویژگی‌ها را بر اساس تعداد فریم‌هایی که ردیابی شده‌اند مرتب می‌کند
		\item اولویت را به ویژگی‌هایی می‌دهد که برای مدت طولانی‌تری ردیابی شده‌اند
		\item اطراف هر ویژگی را در ماسک مسدود می‌کند تا از تشخیص ویژگی‌های جدید در نزدیکی آن جلوگیری شود
	\end{itemize}
	
	این روش باعث می‌شود ویژگی‌ها به صورت یکنواخت در سراسر تصویر توزیع شوند.
	
	
	\section{نتیجه‌گیری و کارهای آینده}
	این گزارش یک شامل دو بخش مرور و اجرا گرفتن علمی از فناوری \gls{VINS}، از مفاهیم هسته‌ای آن تا جزئیات پیاده‌سازی فنی شده است. اگر چه این حوزه بسیار قدیمی است (قدمت آن به ۱۹۵۰ میلادی باز می‌گردد) اما همچنان شاهد تحقیقات فعال در حوزه‌هایی مانند یادگیری عمیق برای استخراج و تطبیق ویژگی، عملیات دائمی با بستن حلقه و قابلیت اطمینان در محیط‌های بسیار چالش‌برانگیز (مانند حرکت با سرعت بالا، نور مستقیم خورشید، شب تاریک) است.
	\subsection{کارهای پژوهشی: کاربرد یادگیری عمیق در  مکان‌یابی}
	در مقاله‌های جدید این حوزه، تمرکز شدیدی بر استفاده از یادگیری عمیق به‌جای استفاده از روش‌های سنتی برای 
	وظیفه‌های مختلف است.
	مثلا در
	\cite{luo2024supervinsrealtimevisualinertialslam}
	پیشنهادهایی بر روی مدل
	\LR{VINS-Fusion}
	با استفاده از ویژگی‌های یادگیری عمیق و روش‌های تطبیق مبتنی بر یادگیری عمیق  برای بهینه‌سازی عملکرد \GLS{SLAM}  ارائه می‌شود.  در این مقاله، یک سیستم 
	\gls{SLAM} تصویری-اینرسی  به نام \LR{SuperVINS} معرفی شده است که برای بهبود پایداری و دقت
	در شرایط ساده مانند نور کم و محو شدن حرکت طراحی شده است. مشکل اصلی سیستم‌های سنتی 
	\gls{SLAM}، از دست دادن ردیابی تصویری در چنین شرایطی است.
	در این مقاله اشاره شده به جای استفاده از روش‌های سنتی مبتنی بر مدل‌سازی هندسی، از مدل‌های عصبی یادگیری عمیق برای تقویت سیستم بینایی استفاده کرده‌اند. دلیل این انتخاب، توانایی شبکه‌های عصبی در استخراج اطلاعات پنهان و ویژگی‌های نامرئی از تصاویر است که توسط ویژگی‌های هندسی ساده قابل شناسایی نیستند \cite{luo2024supervinsrealtimevisualinertialslam}.
	\subsection{بهینه‌سازی و اضافه کردن قابلیت‌های جدید}
	قدم بعدی در این دوره کارآموزی، بهینه‌سازی کدهای منبع موجود و تست آن روی دستگاه‌های موجود باشد. برای بهینه‌سازی کد ردیاب ویژگی، می‌توان مصرف حافظه را با استفاده از استخرهای حافظه و محدود کردن ذخیره تصاویر کاهش داد. محاسبات را می‌توان با روش‌های موازی و SIMD سریع‌تر کرد و الگوریتم‌هایی مثل SIFT برای دقت بهتر استفاده شوند. همچنین، مدیریت خطاها و افزودن لاگ‌های دقیق‌تر می‌تواند عملکرد و عیب‌یابی را بهبود بخشد.\\
	به طور مشخص وظیفه‌ی بعدی من، ویرایش کد منبع
	\LR{VINS-Fusion} به نحوی است که به کمک کارت گرافیک، قابلیت پردازش موزای
	(\LR{SIMD})
	در قسمت ردیابی ویژگی را داشته باشد.
